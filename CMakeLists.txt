cmake_minimum_required(VERSION 3.22)
project(openpak-client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OPENPAK_BUILD_QT "Build the Qt dialogs (openpak::qt)" ON)

include(FetchContent)
# A host that builds its own TLS library (Azahar's bundled LibreSSL) names it OpenSSL::SSL and
# OpenSSL::Crypto first, so one TLS implementation ends up in the binary, not two.
if(NOT TARGET OpenSSL::SSL OR NOT TARGET OpenSSL::Crypto)
  find_package(OpenSSL REQUIRED)
endif()
if(NOT TARGET fmt::fmt)
  find_package(fmt QUIET)
endif()
if(NOT TARGET fmt::fmt AND NOT fmt_FOUND)
  FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.1.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(fmt)
endif()
if(NOT TARGET nlohmann_json::nlohmann_json)
  find_package(nlohmann_json QUIET)
endif()
if(NOT TARGET nlohmann_json::nlohmann_json AND NOT nlohmann_json_FOUND)
  FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(nlohmann_json)
endif()
if(NOT TARGET httplib::httplib)
  find_package(httplib QUIET)
endif()
if(NOT TARGET httplib::httplib AND NOT httplib_FOUND)
  set(HTTPLIB_REQUIRE_OPENSSL ON CACHE BOOL "" FORCE)
  FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(httplib)
endif()

file(GLOB CLIENT_SOURCES CONFIGURE_DEPENDS client/openpak/*.cpp)
add_library(openpak_client STATIC ${CLIENT_SOURCES})
add_library(openpak::client ALIAS openpak_client)
target_include_directories(openpak_client PUBLIC client)
target_link_libraries(openpak_client PUBLIC fmt::fmt nlohmann_json::nlohmann_json httplib::httplib OpenSSL::SSL OpenSSL::Crypto)
target_compile_definitions(openpak_client PUBLIC CPPHTTPLIB_OPENSSL_SUPPORT ENABLE_WEB_SERVICE)
# The NAT check speaks UDP itself.
if(WIN32)
  target_link_libraries(openpak_client PUBLIC ws2_32 iphlpapi)
endif()
# Some hosts (Dolphin) build with -fno-exceptions globally; the library and its
# dependencies throw and catch, so they are re-enabled here no matter what a
# host decided for itself.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(openpak_client PRIVATE -fexceptions)
endif()

# Cloud saves (and the Qt mods page) zip and unzip with libarchive where the host has it, PowerShell on Windows without it,
# and the library's own zip code elsewhere (Android), which needs zlib to read deflated saves.
if(NOT TARGET LibArchive::LibArchive)
  find_package(LibArchive QUIET)
endif()
if(TARGET LibArchive::LibArchive)
  target_link_libraries(openpak_client PUBLIC LibArchive::LibArchive)
  target_compile_definitions(openpak_client PUBLIC OPENPAK_HAVE_LIBARCHIVE)
else()
  message(STATUS "openpak: libarchive not found; cloud saves use PowerShell (Windows) or the built-in zip code")
endif()
if(NOT TARGET ZLIB::ZLIB)
  find_package(ZLIB QUIET)
endif()
if(TARGET ZLIB::ZLIB)
  target_link_libraries(openpak_client PRIVATE ZLIB::ZLIB)
  target_compile_definitions(openpak_client PRIVATE OPENPAK_HAVE_ZLIB)
endif()

if(OPENPAK_BUILD_QT)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  # QtNetwork is optional: a host whose Qt has none (Dolphin's bundled Windows Qt) goes without
  # the chat client, and the Status page times its TCP connect through the client library.
  find_package(Qt6 QUIET COMPONENTS Network)
  set(CMAKE_AUTOMOC ON)
  file(GLOB QT_SOURCES CONFIGURE_DEPENDS qt/openpak/qt/*.cpp qt/openpak/qt/*.h)
  if(NOT TARGET Qt6::Network)
    list(FILTER QT_SOURCES EXCLUDE REGEX "chat_client")
  endif()
  add_library(openpak_qt STATIC ${QT_SOURCES})
  add_library(openpak::qt ALIAS openpak_qt)
  target_include_directories(openpak_qt PUBLIC qt)
  target_link_libraries(openpak_qt PUBLIC openpak::client Qt6::Widgets)
  if(TARGET Qt6::Network)
    target_link_libraries(openpak_qt PUBLIC Qt6::Network)
  else()
    target_compile_definitions(openpak_qt PRIVATE OPENPAK_QT_NO_NETWORK)
  endif()
  # The pages throw and catch too (Dolphin builds with -fno-exceptions globally).
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(openpak_qt PRIVATE -fexceptions)
  endif()

  # OpenPak's translations (UX spec §7), generated from strings/*.json by tools/strings/generate.py
  # and compiled into the library at :/openpak/i18n; openpak::qt::LoadTranslations installs one.
  # Without Qt's LinguistTools the library is English only.
  option(OPENPAK_QT_TRANSLATIONS "Compile the OpenPak translations into openpak_qt" ON)
  if(OPENPAK_QT_TRANSLATIONS)
    find_package(Qt6 QUIET COMPONENTS LinguistTools)
  endif()
  if(OPENPAK_QT_TRANSLATIONS AND COMMAND qt_add_lrelease)
    file(GLOB OPENPAK_TS_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/qt/i18n/openpak_*.ts)
    set_source_files_properties(${OPENPAK_TS_FILES} PROPERTIES OUTPUT_LOCATION ${CMAKE_CURRENT_BINARY_DIR}/openpak_i18n)
    qt_add_lrelease(openpak_qt TS_FILES ${OPENPAK_TS_FILES} QM_FILES_OUTPUT_VARIABLE OPENPAK_QM_FILES)
    qt_add_resources(openpak_qt openpak_i18n PREFIX /openpak/i18n BASE ${CMAKE_CURRENT_BINARY_DIR}/openpak_i18n
                     FILES ${OPENPAK_QM_FILES})
  else()
    message(STATUS "openpak: Qt LinguistTools not found; the OpenPak dialogs are English only")
  endif()
endif()

# Tests: the client's own (standalone builds) and the save-restore guards, which are pure headers
# with no Qt and no libarchive in them, so a build without Qt still runs them.
if(BUILD_TESTING OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  foreach(check baas bcat compatibility contained_path crash_reports friend_structs my_page nat_check network_profile profile_account push session_presence zip_store)
    add_executable(${check}_test tests/${check}_test.cpp)
    target_link_libraries(${check}_test PRIVATE openpak::client)
    add_test(NAME ${check} COMMAND ${check}_test)
  endforeach()
  if(TARGET openpak_qt)
    add_executable(openpak_strings_test tests/strings_test.cpp)
    target_link_libraries(openpak_strings_test PRIVATE openpak::qt)
    if(COMMAND qt_add_lrelease AND OPENPAK_QT_TRANSLATIONS)
      target_compile_definitions(openpak_strings_test PRIVATE OPENPAK_TEST_TRANSLATIONS)
    endif()
    add_test(NAME strings COMMAND openpak_strings_test)
    set_tests_properties(strings PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
  endif()
  foreach(check safe_entry_path restore_swap)
    add_executable(openpak_${check}_test tests/${check}_test.cpp)
    target_include_directories(openpak_${check}_test PRIVATE qt client)
    add_test(NAME ${check} COMMAND openpak_${check}_test)
  endforeach()
endif()
